
class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs) {

        vector<vector<string>> ans;

        for(int i = 0; i < strs.size(); i++) {

            string s = strs[i];
            string temp = s;

            sort(temp.begin(), temp.end());

            bool found = false;

            for(int j = 0; j < ans.size(); j++) {

                string check = ans[j][0];

                sort(check.begin(), check.end());

                if(temp == check) {
                    ans[j].push_back(s);
                    found = true;
                    break;
                }
            }

            if(found == false) {
                vector<string> v;
                v.push_back(s);
                ans.push_back(v);
            }
        }

        return ans;
    }
};
